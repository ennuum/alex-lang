# Commandes de p_cli (testées dans tests/).
from core use len
use io

# Lignes, mots et caractères d'un texte.
func statistiques(texte):
    lignes := 0
    mots := 0
    car := 0
    dans_mot := false
    for c in texte:
        car += 1
        if c == '\n':
            lignes += 1
        if c == ' ' or c == '\n' or c == '\t':
            dans_mot = false
        elif not dans_mot:
            dans_mot = true
            mots += 1
    return [lignes, mots, car]

# Statistiques d'un fichier texte (fins de ligne \r\n et \r ramenées à \n : mêmes comptes sous Windows et Linux).
func compter(fichier):
    return statistiques(io.read_text(fichier)[0])

func executer(args):
    if args["commande"] == "saluer":
        print("Bonjour, " + args["nom"] + " !")
    elif args["commande"] == "compter":
        for f in args["fichiers"]:
            st := compter(f)
            ligne := str(st[0]) + "\t" + str(st[1]) + "\t" + str(st[2]) + "\t" + f
            print(ligne)

# Somme, minimum et maximum d'une liste de nombres non vide.
func resume(valeurs):
    total := 0
    mini := valeurs[0]
    maxi := valeurs[0]
    for v in valeurs:
        total = total + v
        if v < mini:
            mini = v
        if v > maxi:
            maxi = v
    return {"total": total, "min": mini, "max": maxi, "n": len(valeurs)}

# Modèle SIMULÉ du capteur pour la machine simulée de hw (tests dans la VM, sans matériel) :
# io_read / io_write reçoivent le port, la taille (1, 2, 4) et la valeur.
class CapteurSimule:
    func init(base, dixiemes):
        self.base = base
        self.valeur = dixiemes
        self.pret = false
        self.commandes = 0

    func io_write(port, taille, valeur):
        if port == self.base:
            self.commandes += 1
            if valeur == 1:
                self.pret = true
            elif valeur == 2:
                self.pret = false

    func io_read(port, taille):
        if port == self.base + 1:
            if self.pret:
                return 1
            return 0
        if port == self.base + 2:
            self.pret = false
            return self.valeur + 500
        return 255

# alex: native
# p_driver — pilote d'un capteur de température à ports d'entrée-sortie :
#   base + 0 : commande (écriture) : 1 = mesurer, 2 = remettre à zéro
#   base + 1 : état (lecture)      : bit 0 = mesure prête
#   base + 2 : donnée (lecture 16 bits) : température en dixièmes de degré, décalée de +500 (0 = −50,0 °C)
# Compilé dans un noyau (alexc build --native --freestanding) ou exécuté dans la VM sur la machine simulée de hw.
use hw

const CMD_MESURER := 1
const CMD_RAZ := 2

class Capteur:
    base: int := 0
    mesures: int := 0

    func init(base: int):
        self.base = base
        self.mesures = 0

    func remise_a_zero():
        hw.outb(self.base, CMD_RAZ)

    # Lance une mesure et attend le résultat (en dixièmes de degré Celsius).
    func lire_dixiemes() -> int:
        hw.outb(self.base, CMD_MESURER)
        essais := 0
        while (hw.inb(self.base + 1) & 1) == 0:
            essais += 1
            if essais > 100000:
                raise("DriverError", "capteur muet")
        self.mesures += 1
        return hw.inw(self.base + 2) - 500

== ./p_gui/docs/ARCHITECTURE.md
# Architecture de P gui

L'application est découpée en couches ; chacune ne dépend que de celles du dessous.

```
vues/        interface ui.view       → lit l'état, appelle les actions
app/         état et actions          → modifie les modèles, déclenche l'enregistrement
services/    accès au système         → fichiers (stockage.al), réseau, horloge…
modeles/     logique métier           → aucune dépendance à l'interface ni au système
```

## Règles

* **modeles/** : classes et fonctions pures, entièrement testées (`tests/test_taches.al`). Pas d'`use ui.*`,
  pas d'accès aux fichiers.
* **services/** : seul endroit qui touche au système. Écritures atomiques, erreurs remontées en exceptions.
* **app/application.al** : l'unique état de l'application (`etat`) et les actions qui le modifient. Testé sans
  fenêtre (`tests/test_application.al`).
* **vues/** : fonctions qui renvoient des éléments ui.view à partir de l'état ; aucune logique métier. ui.view
  redessine la fenêtre après chaque action.
* **app/config.al** : constantes (nom, version, taille de fenêtre) et emplacements.

## Ajouter une fonctionnalité

1. le comportement dans `modeles/` (et son test) ;
2. l'action correspondante dans `app/application.al` (et son test) ;
3. l'élément d'interface dans `vues/composants.al`, assemblé dans `vues/principale.al`.

## Ressources

`assets/icone.ico` est l'icône de l'exécutable Windows (barre des tâches, explorateur, barre de titre) :
remplacez-la par la vôtre (16, 32, 48 et 256 px). Images et autres ressources vont dans `assets/`.

# Build p_gui for Windows distribution: dist\p_gui.exe (GUI application, icon assets\icone.ico).
#   powershell -ExecutionPolicy Bypass -File scripts\construire.ps1 [-Alexc path\to\alexc.exe]
param([string]$Alexc = "")
$ErrorActionPreference = 'Stop'
$root = Split-Path -Parent $PSScriptRoot
if ($Alexc -eq "") {
    foreach ($name in 'alexc', 'alex') {
        $c = Get-Command $name -ErrorAction SilentlyContinue
        if ($c) { $Alexc = $c.Source; break }
    }

if ($Alexc -eq "") { Write-Error "alexc not found: add it to PATH or pass -Alexc path\to\alexc.exe"; exit 1 }
New-Item -ItemType Directory -Force (Join-Path $root 'dist') | Out-Null
& $Alexc build --native --windows --gui --icon (Join-Path $root 'assets\icone.ico') (Join-Path $root 'src\main.al') -o (Join-Path $root 'dist\p_gui.exe')
exit $LASTEXITCODE

# État et actions de P gui (contrôleur) : les vues lisent `etat` et appellent ces fonctions ; chaque
# modification des tâches est enregistrée. Testé sans fenêtre dans tests/test_application.al.
use modeles.taches as taches
use services.stockage as stockage

etat := {"liste": taches.Liste(), "saisie": "", "filtre": "toutes", "fichier": null, "message": ""}

# Charge les tâches enregistrées (fichier : chemin du JSON, ou null pour ne rien enregistrer).
func demarrer(fichier):
    etat["fichier"] = fichier
    etat["liste"] = stockage.charger(fichier)
    etat["saisie"] = ""
    etat["filtre"] = "toutes"
    etat["message"] = ""

func arreter():
    _enregistrer()

func _enregistrer():
    if etat["fichier"] == null:
        return null
    try:
        stockage.enregistrer(etat["liste"], etat["fichier"])
    catch e:
        etat["message"] = "Enregistrement impossible : " + e.message

func saisir(texte):
    etat["saisie"] = texte

func ajouter():
    if etat["liste"].ajouter(etat["saisie"]):
        etat["saisie"] = ""
        etat["message"] = ""
        _enregistrer()
    else:
        etat["message"] = "Le titre de la tâche est vide."

func basculer(i):
    etat["liste"].basculer(i)
    _enregistrer()

func supprimer(i):
    etat["liste"].supprimer(i)
    _enregistrer()

func vider_faites():
    n := etat["liste"].vider_faites()
    etat["message"] = str(n) + " tâche(s) terminée(s) retirée(s)."
    _enregistrer()

func filtrer(nom):
    etat["filtre"] = nom

# Modèle de P gui : la liste de tâches, sans interface ni fichier (testé dans tests/test_taches.al).
from core use len, type
from collections use append, contains_key
use string

const FILTRES := ["toutes", "a_faire", "faites"]

class Liste:
    func init():
        self.elements = []

    # Ajoute une tâche ; refuse un titre vide. Renvoie true si ajoutée.
    func ajouter(titre):
        t := string.trim(titre)
        if t == "":
            return false
        append(self.elements, {"titre": t, "faite": false})
        return true

    func basculer(i):
        if i >= 0 and i < len(self.elements):
            self.elements[i]["faite"] = not self.elements[i]["faite"]

    func supprimer(i):
        garde := []
        k := 0
        for t in self.elements:
            if k != i:
                append(garde, t)
            k += 1
        self.elements = garde

    # Retire les tâches terminées ; renvoie leur nombre.
    func vider_faites():
        garde := []
        for t in self.elements:
            if not t["faite"]:
                append(garde, t)
        n := len(self.elements) - len(garde)
        self.elements = garde
        return n

    # Nombre de tâches non faites.
    func restantes():
        n := 0
        for t in self.elements:
            if not t["faite"]:
                n += 1
        return n

    # Indices des tâches visibles pour un filtre : "toutes", "a_faire" ou "faites".
    func visibles(filtre):
        out := []
        i := 0
        for t in self.elements:
            if filtre == "toutes" or (filtre == "faites") == t["faite"]:
                append(out, i)
            i += 1
        return out

    # Données sérialisables (liste de dicts), pour le stockage.
    func vers_donnees():
        out := []
        for t in self.elements:
            append(out, {"titre": t["titre"], "faite": t["faite"]})
        return out

# Liste reconstruite depuis des données lues (entrées invalides ignorées).
func depuis_donnees(donnees):
    l := Liste()
    if type(donnees) != "list":
        return l
    for e in donnees:
        if type(e) == "dict" and contains_key(e, "titre") and type(e["titre"]) == "str" and string.trim(e["titre"]) != "":
            append(l.elements, {"titre": string.trim(e["titre"]), "faite": contains_key(e, "faite") and e["faite"] == true})
    return l

const VGA := 753664 # 0xB8000 : écran texte 80 x 25

func ecran(ligne: int, s: str, attr: int):
    i := 0
    for c in s:
        hw.poke16(VGA + 2 * (ligne * 80 + i), texte.code(c) + (attr << 8))

from core use len
from collections use append
use string

# Version de la bibliothèque.
const VERSION := "0.1.0"

# Identifiant d'URL : minuscules, lettres et chiffres, mots séparés par « - ».
func slug(texte):
    out := []
    tiret := false
    for c in string.lower(texte):
        ok := (c >= 'a' and c <= 'z') or (c >= '0' and c <= '9')
        if ok:
            if tiret and len(out) > 0:
                append(out, "-")
            append(out, str(c))
            tiret = false
        else:
            tiret = true
    return string.join("", out)

# Nombre de mots (séparés par des espaces).
func mots(texte):
    n := 0
    dans_mot := false
    for c in texte:
        if c == ' ' or c == '\t' or c == '\n':
            dans_mot = false
        elif not dans_mot:
            dans_mot = true
            n += 1
    return n

# Routes de p_server : fonctions pures (méthode, chemin, paramètres) -> [statut, corps, "json" | "text"], et leur
# traduction en réponse HTTP (traiter). Les deux se testent sans réseau.
from collections use contains_key
use network.http as http

compteur := {"visites": 0}

func repondre(methode, chemin, params):
    compteur["visites"] += 1
    if methode == "GET" and chemin == "/":
        return [200, "P server fonctionne.\n", "text"]
    if methode == "GET" and chemin == "/api/bonjour":
        nom := "monde"
        if contains_key(params, "nom"):
            nom = params["nom"]
        return [200, {"message": "Bonjour, " + nom + " !"}, "json"]
    if methode == "GET" and chemin == "/api/stats":
        return [200, {"visites": compteur["visites"]}, "json"]
    return [404, "introuvable : " + chemin + "\n", "text"]

# Réponse HTTP (network.http.Response) d'une requête : JSON ou texte, avec le statut de repondre.
func traiter(methode, chemin, params):
    r := repondre(methode, chemin, params)
    if r[2] == "json":
        return http.json_response(r[1], r[0])
    return http.text(r[1], r[0])

from core use len
use io

# Lignes, mots et caractères d'un texte.
func statistiques(texte):
    lignes := 0
    mots := 0
    car := 0
    dans_mot := false
    for c in texte:
        car += 1
        if c == '\n':
            lignes += 1
        if c == ' ' or c == '\n' or c == '\t':
            dans_mot = false
        elif not dans_mot:
            dans_mot = true
            mots += 1
    return [lignes, mots, car]

# Statistiques d'un fichier texte (fins de ligne \r\n et \r ramenées à \n : mêmes comptes sous Windows et Linux).
func compter(fichier):
    return statistiques(io.read_text(fichier)[0])

func executer(args):
    if args["commande"] == "saluer":
        print("Bonjour, " + args["nom"] + " !")
    elif args["commande"] == "compter":
        for f in args["fichiers"]:
            st := compter(f)
            ligne := str(st[0]) + "\t" + str(st[1]) + "\t" + str(st[2]) + "\t" + f
            print(ligne)
